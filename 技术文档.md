# AI Agent 工具调用平台 — 技术设计文档

> 项目定位：一个支持自定义工具注册、多模型调度、多轮对话的智能 Agent 平台，用户可以通过自然语言指令让 Agent 自动选择并调用外部工具完成复杂任务。

---

## 一、项目背景与价值

### 1.1 为什么做这个项目？

大模型本身只能生成文本，无法直接操作外部世界。Agent（智能体）的核心能力是**让大模型具备"动手"的能力** —— 通过工具调用（Tool Use / Function Calling）让模型查询数据库、调用 API、执行代码、操作文件系统。

这个项目展示了：
- 对 **Agent 架构** 的深入理解（ReAct、Tool Use、Multi-Agent）
- 对 **大模型能力边界** 的认知（模型擅长推理，工具擅长执行）
- 对 **工程化落地** 的能力（不是玩具 Demo，而是可扩展的平台）

### 1.2 面试话术

> "我做了一个 Agent 工具调用平台，支持自定义工具注册、多模型调度、多轮对话状态管理。核心设计是把模型的推理能力和工具的执行能力解耦，通过标准化的工具接口让 Agent 可以灵活扩展能力边界。最新版本实现了真正的流式输出，支持 token 级别的实时响应。"

### 3.1 工具注册与管理系统

工具是 Agent 的"手脚"，需要标准化的接口让模型知道有哪些工具可用、怎么调用。

#### 3.1.1 工具描述规范（Tool Schema）

```python
from pydantic import BaseModel, Field
from typing import Any, Optional
from enum import Enum

class ParameterType(str, Enum):
    STRING = "string"
    INTEGER = "integer"
    FLOAT = "float"
    BOOLEAN = "boolean"
    ARRAY = "array"
    OBJECT = "object"

class ToolParameter(BaseModel):
    """工具参数定义"""
    name: str
    type: ParameterType = ParameterType.STRING
    description: str = ""
    required: bool = False
    default: Any = None
    enum: list[str] | None = None

class ToolDefinition(BaseModel):
    """工具完整定义"""
    name: str                            # 唯一标识，如 "http_get"
    display_name: str = ""               # 显示名称
    description: str                     # 工具描述（模型通过这个理解工具用途）
    category: str = "general"            # 分类，如 "网络"、"文件"、"数据"
    parameters: list[ToolParameter] = [] # 参数列表
    enabled: bool = True
    timeout: int = 30                    # 超时时间（秒）

class ToolResult(BaseModel):
    """工具执行结果"""
    success: bool
    data: Any = None
    error: str | None = None
    execution_time: float = 0.0
```

#### 3.1.2 工具注册机制

```python
class ToolRegistry:
    """工具注册中心 — 管理所有可用工具"""

    def __init__(self):
        self._tools: dict[str, ToolDefinition] = {}
        self._executors: dict[str, Callable] = {}

    def register(self, name: str, description: str, category: str = "general",
                 parameters: list[ToolParameter] | None = None, timeout: int = 30):
        """装饰器：注册一个工具"""
        def decorator(func: Callable):
            tool_params = parameters or self._infer_parameters(func)
            tool_def = ToolDefinition(
                name=name,
                display_name=name.replace("_", " ").title(),
                description=description,
                category=category,
                parameters=tool_params,
                timeout=timeout,
            )
            self._tools[name] = tool_def
            self._executors[name] = func
            return func
        return decorator

    def _infer_parameters(self, func: Callable) -> list[ToolParameter]:
        """从函数签名自动推断参数定义"""
        sig = inspect.signature(func)
        params = []
        type_map = {
            str: ParameterType.STRING,
            int: ParameterType.INTEGER,
            float: ParameterType.FLOAT,
            bool: ParameterType.BOOLEAN,
            list: ParameterType.ARRAY,
            dict: ParameterType.OBJECT,
        }
        for name, param in sig.parameters.items():
            if name in ("self", "cls"):
                continue
            required = param.default is inspect.Parameter.empty
            params.append(ToolParameter(
                name=name,
                type=type_map.get(param.annotation, ParameterType.STRING),
                description=f"参数: {name}",
                required=required,
                default=None if required else param.default,
            ))
        return params

    async def execute(self, name: str, arguments: dict) -> ToolResult:
        """执行工具"""
        executor = self._executors.get(name)
        if not executor:
            return ToolResult(success=False, error=f"工具 '{name}' 不存在")

        start = time.time()
        try:
            if inspect.iscoroutinefunction(executor):
                result = await executor(**arguments)
            else:
                result = executor(**arguments)
            return ToolResult(success=True, data=result, execution_time=time.time() - start)
        except Exception as e:
            return ToolResult(success=False, error=str(e), execution_time=time.time() - start)
```

#### 3.1.3 动态工具发现与热加载

```python
class DynamicToolLoader:
    """动态工具加载器 — 支持从文件系统热加载工具"""

    def __init__(self, registry: ToolRegistry, tools_dir: str = "./tools"):
        self.registry = registry
        self.tools_dir = Path(tools_dir)

    def discover_and_load(self):
        """扫描工具目录，自动加载所有工具模块"""
        if not self.tools_dir.exists():
            return
        for py_file in self.tools_dir.glob("*.py"):
            if py_file.name.startswith("_"):
                continue
            self._load_module(py_file)

    def _load_module(self, file_path: Path):
        """加载单个工具模块"""
        module_name = f"tools.{file_path.stem}"
        spec = importlib.util.spec_from_file_location(module_name, file_path)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
```

### 3.2 Agent 核心引擎

#### 3.2.1 ReAct 循环实现

Agent 的核心是一个 **思考-行动-观察（Think-Act-Observe）** 循环：

```
用户输入 → [思考：需要做什么？] → [行动：调用工具] → [观察：工具结果]
     ↑                                                    │
     └────────────── 继续思考或返回最终答案 ←──────────────┘
```

```python
class AgentOrchestrator:
    """Agent 编排器 — 实现 Think-Act-Observe 循环，支持真正的流式输出"""

    def __init__(self, model_router: ModelRouter, tool_registry: ToolRegistry, config: dict):
        self.router = model_router
        self.tools = tool_registry
        self.config = config.get("agent", {})
        self.max_steps = self.config.get("max_steps", 10)
        self.db = None  # 用于记录工具调用日志

    async def run(self, user_input: str, conversation_id: str = None,
                  model: str = None, history: list[Message] = None) -> list[Message]:
        """同步执行 ReAct 循环"""
        model = model or self.config.get("default_model", "deepseek-chat")
        messages = []

        # 系统提示
        system_content = self._build_system_prompt()
        messages.append({"role": "system", "content": system_content})

        # 历史消息
        if history:
            for msg in history[-20:]:
                if msg.role == "user":
                    messages.append({"role": "user", "content": msg.content})
                elif msg.role == "assistant":
                    messages.append({"role": "assistant", "content": msg.content})

        # 用户输入
        messages.append({"role": "user", "content": user_input})

        # 获取工具定义
        tool_schemas = self.tools.to_function_calls()

        # ReAct 循环
        for step in range(self.max_steps):
            response = await self.router.chat(
                messages=messages, model=model,
                tools=tool_schemas if tool_schemas else None,
            )

            if response.tool_calls:
                # 执行工具
                for tool_call in response.tool_calls:
                    func = tool_call["function"]
                    tool_name = func["name"]
                    tool_args = json.loads(func["arguments"])

                    result = await self.tools.execute(tool_name, tool_args)

                    # 记录日志
                    if self.db and conversation_id:
                        self.db.log_tool_call(
                            conversation_id=conversation_id,
                            tool_name=tool_name,
                            arguments=tool_args,
                            result=result.model_dump(),
                            success=result.success,
                            execution_time=result.execution_time,
                        )

                    # 加入上下文继续循环
                    messages.append({"role": "tool", "tool_call_id": tool_call["id"],
                                    "content": json.dumps(result.model_dump())})
            else:
                # 返回最终答案
                return [Message(role="assistant", content=response.content,
                               metadata={"steps": step + 1, "model": response.model})]
```

```python
async def chat_stream(self, messages, model, tools=None, temperature=0.7, max_tokens=4096):
    """DeepSeek 真正的流式输出"""
    body = {
        "model": model,
        "messages": messages,
        "temperature": temperature,
        "max_tokens": max_tokens,
        "stream": True,
    }
    if tools:
        body["tools"] = tools

    async with httpx.AsyncClient(timeout=120) as client:
        async with client.stream("POST", f"{self.base_url}/chat/completions",
                                 json=body, headers=headers) as resp:
            async for line in resp.aiter_lines():
                if not line.startswith("data: "):
                    continue
                data_str = line[6:]
                if data_str.strip() == "[DONE]":
                    break

                chunk = json.loads(data_str)
                delta = chunk["choices"][0].get("delta", {})

                if "content" in delta and delta["content"]:
                    yield {"type": "token", "content": delta["content"]}

                if "tool_calls" in delta:

                    ...

    yield {"type": "done", "response": response.model_dump()}
```

#### 3.2.3 模型路由器

```python
class ModelRouter:
    """多模型路由器 — 支持多模型调度、故障转移"""

    def __init__(self, config: dict):
        self.adapters: dict[str, BaseModelAdapter] = {}
        self.model_map: dict[str, str] = {}  # model_name -> adapter_name

    async def chat(self, messages, model="deepseek-chat", tools=None, **kwargs):
        """同步对话"""
        adapter_name, adapter = self._get_adapter(model)
        try:
            return await adapter.chat(messages=messages, model=model, tools=tools, **kwargs)
        except Exception as e:
            # 故障转移
            for name, fallback in self.adapters.items():
                if name != adapter_name:
                    try:
                        return await fallback.chat(messages=messages, model=model, tools=tools, **kwargs)
                    except Exception:
                        continue
            raise RuntimeError(f"所有模型均不可用: {e}")

    async def chat_stream(self, messages, model="deepseek-chat", tools=None, **kwargs):
        """流式对话 — 返回 token 级别的流"""
        adapter_name, adapter = self._get_adapter(model)
        async for chunk in adapter.chat_stream(messages=messages, model=model, tools=tools, **kwargs):
            yield chunk
```

```python
async def chat_stream(request: ChatRequest):
    """真正的流式对话接口（SSE）— 逐 token 输出"""
    async def event_generator():
        async for msg in orchestrator.run_stream(
            user_input=request.message,
            conversation_id=conv_id,
            model=request.model,
        ):
            msg_type = msg.metadata.get("type", "")
            if msg_type == "token":
                yield f"data: {json.dumps({'type': 'token', 'content': msg.content})}\n\n"
            elif msg_type == "tool_call":
                yield f"data: {json.dumps({'type': 'tool_call', 'tool_call': msg.tool_call})}\n\n"
            elif msg_type == "tool_result":
                yield f"data: {json.dumps({'type': 'tool_result', 'tool_result': msg.tool_result})}\n\n"
            elif msg_type == "done":
                yield f"data: {json.dumps({'type': 'done', 'metadata': msg.metadata})}\n\n"
        yield "data: [DONE]\n\n"

    return StreamingResponse(event_generator(), media_type="text/event-stream")
```

```python
async def generate_title(self, user_input: str, model: str = None) -> str:
    response = await self.router.chat(
        messages=[

            {"role": "user", "content": user_input},
```
